# AIccounting 软件配置与运维文档

## 一、本地环境配置

### 1. 系统要求

AIccounting项目是一个本地化的财务管理系统，支持跨平台运行。系统主要组件包括：
- Web后端：Python 3.11 + Flask + MySQL
- Web前端：Vue.js 2.x + webpack
- 移动端：Vue 3 + Cordova（Android）

**操作系统支持：**
- Windows 10/11（推荐）
- macOS 10.15+
- Ubuntu 18.04+

### 2. 环境依赖安装

#### Python环境配置
```bash
# 安装Python 3.11
python --version  # 确认版本为3.11+

# 创建虚拟环境
python -m venv venv

# Windows激活虚拟环境
venv\Scripts\activate
# macOS/Linux激活虚拟环境
source venv/bin/activate

# 安装后端依赖
pip install -r requirements.txt
```

#### Node.js环境配置
```bash
# 安装Node.js 16+
node --version
npm --version

# 安装Web前端依赖
cd App
npm install

# 安装移动端依赖
cd ../App-Mobile
npm install
```

#### MySQL数据库配置
```bash
# 安装MySQL 8.0
mysql --version

# 创建数据库
mysql -u root -p
CREATE DATABASE aiccounting CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
CREATE USER 'aiccounting'@'localhost' IDENTIFIED BY 'your_password';
GRANT ALL PRIVILEGES ON aiccounting.* TO 'aiccounting'@'localhost';
FLUSH PRIVILEGES;
```

## 四、数据备份与恢复

### 1. 数据库备份

#### 自动备份脚本
创建`backup.sh`（Linux/macOS）或`backup.bat`（Windows）：
```bash
#!/bin/bash
# 数据库备份脚本
BACKUP_DIR="./backups"
DATE=$(date +%Y%m%d_%H%M%S)
BACKUP_FILE="aiccounting_backup_$DATE.sql"

mkdir -p $BACKUP_DIR
mysqldump -u aiccounting -p aiccounting > "$BACKUP_DIR/$BACKUP_FILE"
echo "数据库备份完成: $BACKUP_FILE"

# 保留最近7天的备份
find $BACKUP_DIR -name "*.sql" -mtime +7 -delete
```

#### 手动备份
```bash
# 导出数据库
mysqldump -u aiccounting -p aiccounting > backup.sql

# 导出特定表
mysqldump -u aiccounting -p aiccounting transactions > transactions_backup.sql
```

### 2. 数据恢复

```bash
# 从备份文件恢复
mysql -u aiccounting -p aiccounting < backup.sql

# 恢复特定表
mysql -u aiccounting -p aiccounting < transactions_backup.sql
```

### 3. 移动端数据备份

```javascript
// 导出SQLite数据
async function exportData() {
  const db = window.sqlitePlugin.openDatabase({name: 'aiccounting.db'});
  
  return new Promise((resolve, reject) => {
    db.transaction(tx => {
      tx.executeSql('SELECT * FROM transactions', [], (tx, results) => {
        const data = Array.from(results.rows);
        const jsonData = JSON.stringify(data, null, 2);
        
        // 保存到本地文件
        window.resolveLocalFileSystemURL(cordova.file.documentsDirectory, 
          dirEntry => {
            dirEntry.getFile('backup.json', {create: true}, fileEntry => {
              fileEntry.createWriter(writer => {
                writer.write(new Blob([jsonData], {type: 'application/json'}));
                resolve('备份成功');
              });
            });
          }
        );
      });
    });
  });
}
```

## 五、本地监控与日志管理

### 1. 系统监控

#### 服务状态监控
创建`monitor.py`脚本：
```python
import psutil
import logging
from datetime import datetime

def check_system_status():
    """检查系统状态"""
    status = {
        'timestamp': datetime.now().isoformat(),
        'cpu_percent': psutil.cpu_percent(interval=1),
        'memory_percent': psutil.virtual_memory().percent,
        'disk_usage': psutil.disk_usage('/').percent,
        'python_processes': len([p for p in psutil.process_iter() if 'python' in p.name().lower()])
    }
    
    logging.info(f"系统状态: {status}")
    
    # 检查异常情况
    if status['cpu_percent'] > 80:
        logging.warning(f"CPU使用率过高: {status['cpu_percent']}%")
    if status['memory_percent'] > 80:
        logging.warning(f"内存使用率过高: {status['memory_percent']}%")
    
    return status

if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO, 
                       format='%(asctime)s - %(levelname)s - %(message)s')
    check_system_status()
```

#### 数据库监控
```python
import pymysql
import logging

def check_database_status():
    """检查数据库状态"""
    try:
        connection = pymysql.connect(
            host='localhost',
            user='aiccounting',
            password='your_password',
            database='aiccounting'
        )
        
        with connection.cursor() as cursor:
            # 检查连接数
            cursor.execute("SHOW STATUS LIKE 'Threads_connected'")
            threads = cursor.fetchone()[1]
            
            # 检查表大小
            cursor.execute("""
                SELECT table_name, 
                       ROUND(((data_length + index_length) / 1024 / 1024), 2) as size_mb
                FROM information_schema.tables 
                WHERE table_schema = 'aiccounting'
            """)
            tables = cursor.fetchall()
            
            logging.info(f"数据库连接数: {threads}")
            for table in tables:
                logging.info(f"表 {table[0]} 大小: {table[1]} MB")
                
        connection.close()
        return True
        
    except Exception as e:
        logging.error(f"数据库连接失败: {e}")
        return False
```

### 2. 日志管理

#### 后端日志配置
在`main.py`中配置日志：
```python
import logging
from logging.handlers import RotatingFileHandler
import os

def setup_logging():
    """配置日志系统"""
    if not os.path.exists('logs'):
        os.makedirs('logs')
    
    # 配置文件日志
    file_handler = RotatingFileHandler(
        'logs/aiccounting.log', 
        maxBytes=10*1024*1024,  # 10MB
        backupCount=5
    )
    file_handler.setFormatter(logging.Formatter(
        '%(asctime)s %(levelname)s: %(message)s [in %(pathname)s:%(lineno)d]'
    ))
    file_handler.setLevel(logging.INFO)
    
    # 配置控制台日志
    console_handler = logging.StreamHandler()
    console_handler.setFormatter(logging.Formatter(
        '%(asctime)s - %(name)s - %(levelname)s - %(message)s'
    ))
    
    # 应用配置
    app.logger.addHandler(file_handler)
    app.logger.addHandler(console_handler)
    app.logger.setLevel(logging.INFO)
    app.logger.info('AIccounting 启动')

if __name__ == '__main__':
    setup_logging()
    app.run(debug=True, port=5000)
```

## 六、常见故障处理

### 1. 数据库连接问题

**问题症状**：
- 应用启动时报"Can't connect to MySQL server"
- 数据无法保存或读取

**解决方案**：
```bash
# 检查MySQL服务状态
systemctl status mysql      # Linux
brew services list | grep mysql  # macOS
net start mysql             # Windows

# 重启MySQL服务
systemctl restart mysql     # Linux
brew services restart mysql # macOS
net stop mysql && net start mysql  # Windows

# 检查端口占用
netstat -an | grep 3306

# 检查用户权限
mysql -u root -p
SHOW GRANTS FOR 'aiccounting'@'localhost';
```

### 2. 端口冲突问题

**问题症状**：
- "Port 5000 is already in use"
- "EADDRINUSE: address already in use"

**解决方案**：
```bash
# 查找占用端口的进程
netstat -ano | findstr :5000    # Windows
lsof -i :5000                   # macOS/Linux

# 终止进程
taskkill /PID <pid> /F          # Windows
kill -9 <pid>                  # macOS/Linux

# 或修改配置使用其他端口
export FLASK_PORT=5001
```

### 3. 依赖安装问题

**问题症状**：
- npm install失败
- pip install报错
- 模块导入错误

**解决方案**：
```bash
# 清理npm缓存
npm cache clean --force
rm -rf node_modules package-lock.json
npm install

# 清理pip缓存
pip cache purge
pip install --upgrade pip
pip install -r requirements.txt --force-reinstall

# Python虚拟环境重建
deactivate
rm -rf venv
python -m venv venv
source venv/bin/activate
pip install -r requirements.txt
```

### 4. 移动端构建问题

**问题症状**：
- Cordova构建失败
- APK生成错误
- 插件安装失败

**解决方案**：
```bash
# 清理Cordova缓存
cordova clean android

# 重新添加平台
cordova platform remove android
cordova platform add android

# 检查Android SDK配置
echo $ANDROID_HOME
echo $JAVA_HOME

# 更新插件
cordova plugin remove cordova-sqlite-storage
cordova plugin add cordova-sqlite-storage
```

import time
import functools

def performance_monitor(func):
    @functools.wraps(func)
    def wrapper(*args, **kwargs):
        start_time = time.time()
        result = func(*args, **kwargs)
        end_time = time.time()
        
        execution_time = end_time - start_time
        if execution_time > 1.0:  # 超过1秒记录警告
            app.logger.warning(
                f"慢查询警告: {func.__name__} 执行时间: {execution_time:.2f}秒"
            )
        
        return result
    return wrapper

// 数据缓存
const cache = new Map();

function getCachedData(key, fetchFunc) {
  if (cache.has(key)) {
    return Promise.resolve(cache.get(key));
  }
  
  return fetchFunc().then(data => {
    cache.set(key, data);
