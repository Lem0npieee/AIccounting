<template>
  <div class="tab-bar">
    <div 
      v-for="(item, index) in tabs" 
      :key="index" 
      class="tab-item"
      :class="{ active: $route.path === item.path }"
      @click="navigateTo(item.path)"
    >
      <div class="icon">
        <i :class="item.icon"></i>
      </div>
      <div class="tab-title">{{ item.title }}</div>
    </div>
  </div>
</template>
 
<script>
export default {
  name: 'TabBar',
  data() {
    return {
      tabs: [
        {
          title: '明细',
          path: '/details',
          icon: 'icon-details'
        },
        {
          title: '记账',
          path: '/record',
          icon: 'icon-record'
        },
        {
          title: '图表',
          path: '/chart',
          icon: 'icon-chart'
        }
      ]
    }
  },
  methods: {
    navigateTo(path) {
      // 避免NavigationDuplicated错误
      if (this.$route.path !== path) {
        this.$router.push(path).catch(() => {})
      }
    }
  }
}
</script>
